def extract_date_from_filename(filename):
    """
    Extract date from filename using a pattern like YYYYMMDD.
    Returns the date as a string, or None if not found.

    Args:
        filename: The filename to extract date from.

    Returns:
        Date string (YYYYMMDD) or None.
    """
    # Pattern to match YYYYMMDD format
    pattern = r'(\d{4})(\d{2})(\d{2})'
    match = re.search(pattern, filename)
    if match:
        return match.group(0)
    return None


def list_csv_files_with_dates(model_dir):
    """
    List all CSV files in a model directory that contain a date in their filename.

    Args:
        model_dir: Path to the model directory.

    Returns:
        List of tuples (filepath, extracted_date).
    """
    if not os.path.exists(model_dir):
        return []

    files_with_dates = []
    for filename in os.listdir(model_dir):
        if filename.endswith('.csv'):
            date = extract_date_from_filename(filename)
            if date:
                filepath = os.path.join(model_dir, filename)
                files_with_dates.append((filepath, date))

    return files_with_dates

def compute_mae(predictions, ground_truth):
    """
    Compute Mean Absolute Error between predictions and ground truth.
    Both should be pandas Series with the same index.

    Args:
        predictions: Series of predicted values.
        ground_truth: Series of true values.

    Returns:
        MAE as a float, or None if computation fails.
    """
    try:
        # Align the series
        aligned = pd.concat([predictions, ground_truth], axis=1, join='inner')
        if len(aligned) == 0:
            return None
        mae = (aligned.iloc[:, 0] - aligned.iloc[:, 1]).abs().mean()
        return mae
    except Exception as e:
        logger.error(f"Error computing MAE: {e}")
        return None
