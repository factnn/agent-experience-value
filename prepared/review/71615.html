- owner

- ref
- filePath (the portion after the ref)

- branches.txt: each line has "<branch> <hash>"
- tags.txt: each line has "<tag> <hash>"

Behavior in offline mode:
- If the ref segment after /tree/ is not HEAD (or is missing) and a mapping exists for the ref in branches.txt or tags.txt, resolve the ref to the mapped commit hash.
- If the ref is HEAD or missing, use HEAD as the ref.
- If the ref is not found in mappings, treat it as a literal ref (do not fail).

Edge cases:
- If the URL path is /owner/repo/tree/HEAD/path/to/file, the filePath should be "path/to/file".
- If the path is /owner/repo/tree/branch-name/path/to/file and branch-name is in branches.txt, the ref should resolve to the corresponding commit hash. If not found, the ref remains branch-name.
- If the path is /owner/repo/tree/v1.0/path and v1.0 is in tags.txt, the ref should resolve to the corresponding commit hash. If not found, the ref remains v1.0.
- If there is no path after the ref, /owner/repo/tree, the filePath should be "".

#!/usr/bin/env python3
import sys
import json
import os

def load_mappings(filepath):
    """Load branch/tag to hash mappings from a file."""
    mappings = {}
    if os.path.exists(filepath):
        with open(filepath, 'r') as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith('#'):
                    parts = line.split(None, 1)
                    if len(parts) == 2:
                        mappings[parts[0]] = parts[1]
    return mappings

def resolve_ref(ref, branches, tags):
    """Resolve a ref to its hash if found in mappings."""
    if ref == 'HEAD':
        return 'HEAD'
    if ref in branches:
        return branches[ref]
    if ref in tags:
        return tags[ref]
    return ref

def main():
    args = sys.argv[1:]
    if not args:
        print("Usage: parse_tree_url.py /owner/repo/tree/<ref>[/path] [--offline]", file=sys.stderr)
        sys.exit(2)

    path = args[0]
    offline = '--offline' in args

    parts = path.strip('/').split('/', 4)
    owner = parts[0] if len(parts) > 0 else ''
    repo = parts[1] if len(parts) > 1 else ''

    ref = 'HEAD'
    file_path = ''

    if len(parts) >= 3 and parts[2] == 'tree':
        if len(parts) >= 4:
            ref = parts[3] or 'HEAD'
        if len(parts) == 5:
            file_path = parts[4]
    else:
        # Invalid format, but handle gracefully
        pass

    resolved_ref = ref
    if offline:
        script_dir = os.path.dirname(os.path.abspath(__file__))
        branches = load_mappings(os.path.join(script_dir, 'branches.txt'))
        tags = load_mappings(os.path.join(script_dir, 'tags.txt'))
        resolved_ref = resolve_ref(ref, branches, tags)

    output = {
        "pageType": "TREE",
        "owner": owner,
        "repo": repo,
        "ref": resolved_ref,
        "filePath": file_path
    }

    print(json.dumps(output, separators=(',', ':')))

if __name__ == '__main__':
    main()
