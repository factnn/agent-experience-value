"""Django system check framework."""

class CheckMessage:
    def __init__(self, level, msg, hint=None, obj=None, id=None):
        self.level = level
        self.msg = msg
        self.hint = hint
        self.obj = obj
        self.id = id

    def __repr__(self):
        return f"{self.__class__.__name__}({self.msg})"


class Error(CheckMessage):
    level = 40

    def __init__(self, msg, hint=None, obj=None, id=None):
        super().__init__(self.level, msg, hint=hint, obj=obj, id=id)


# Registry for checks
_checks = []

def register(check=None, **kwargs):
    """Register a check function."""
    def decorator(check_func):
        _checks.append((check_func, kwargs))
        return check_func

    if check is None:
        return decorator
    else:
        _checks.append((check, kwargs))
        return check


def run_checks(app_configs=None, **kwargs):
    """Run all registered checks and return list of Errors."""
    errors = []
    for check_func, check_kwargs in _checks:
        errors.extend(check_func(app_configs=app_configs, **kwargs, **check_kwargs))
    return errors

class URLResolver:

        self.pattern = pattern
        self.url_patterns = url_patterns or []
        self.default_args = default_args or {}
