Patches for Sphinx directives.

This module contains patched versions of docutils directives that are
modified to work better with Sphinx.
"""

import warnings
from typing import List, Tuple

from docutils import nodes
from docutils.parsers.rst import directives as rst_directives, tables
from docutils.utils import system_message

from sphinx.util import logging
from sphinx.util.nodes import set_source_info
from sphinx.deprecation import RemovedInSphinx60Warning

logger = logging.getLogger(__name__)


class CSVTable(tables.CSVTable):
    """The csv-table directive which sets source and line information to its caption.

    This patched version treats absolute paths in :file: as relative to the
    Sphinx source directory (env.srcdir), mirroring the behavior of the
    figure directive.

    Only for docutils-0.13 or older version."""

    option_spec = tables.CSVTable.option_spec.copy()

    def get_source_info(self, env):
        """Get the source directory and current document directory."""
        srcdir = env.srcdir
        docdir = os.path.dirname(env.doc2path(env.docname))
        return srcdir, docdir

    def resolve_file_path(self, filename: str) -> str:
        """Resolve the file path for the csv-file option.

        If the path is absolute (starts with / or a drive letter on Windows),
        treat it as relative to the source directory. Otherwise, keep it as
        a relative path from the current document.
        """
        try:
            env = self.state.document.settings.env
        except AttributeError:
            # No Sphinx environment, return path as-is
            return filename

        # Check if the path is absolute (starts with / on Unix or drive letter on Windows)
        is_absolute = os.path.isabs(filename)

        if is_absolute:
            # Treat absolute path as relative to source directory
            # Strip the leading slash/drive and make it relative
            if filename.startswith('/'):
                # Unix-style absolute path
                relative_path = filename.lstrip('/')
            elif len(filename) >= 2 and filename[1] == ':':
                # Windows-style absolute path (e.g., C:\path)
                relative_path = filename[2:].lstrip('\\/')
            else:
                relative_path = filename.lstrip('/\\')

            # Get the source directory
            srcdir = env.srcdir

            # Construct the full path from source directory
            full_path = os.path.join(srcdir, relative_path)

            # Get the current document's directory
            docdir = os.path.dirname(env.doc2path(env.docname))

            # Make the path relative to the current document
            try:
                # Try to make it relative to the document directory
                resolved = os.path.relpath(full_path, docdir)
            except ValueError:
                # On Windows, if paths are on different drives, can't make relative
                resolved = relative_path

            # Issue a warning about the path normalization
            logger.warning(
                'csv-table :file: path %s is being interpreted as relative to '
                'the source directory (resolved to %s).',
                filename, resolved,
                location=self.state_machine.get_source_and_line(self.lineno)
            )

            return resolved

        # Relative path - return as-is
        return filename

    def run(self) -> List[nodes.Node]:
        warnings.warn('CSVTable is deprecated.',
                      RemovedInSphinx60Warning)

        # Handle the file option if present
        if 'file' in self.options:
            original_file = self.options['file']
            resolved_file = self.resolve_file_path(original_file)
            self.options['file'] = resolved_file

        return super().run()

    def make_title(self) -> Tuple[nodes.title, List[system_message]]:
        title, message = super().make_title()
        if title:
            set_source_info(self, title)
        return title, message

from sphinx.util.nodes import set_source_info
from sphinx.deprecation import RemovedInSphinx60Warning

logger = logging.getLogger(__name__)


class CSVTable(BaseCSVTable):
    """The csv-table directive which sets source and line information to its caption.

    This patched version treats absolute paths in :file: as relative to the
    Sphinx source directory (env.srcdir), mirroring the behavior of the
    figure directive.

    Only for docutils-0.13 or older version."""

    option_spec = BaseCSVTable.option_spec.copy()

    def resolve_file_path(self, filename: str) -> str:
        """Resolve the file path for the csv-file option.

        If the path is absolute (starts with / or a drive letter on Windows),
        treat it as relative to the source directory. Otherwise, keep it as
        a relative path from the current document.
        """
        try:
            env = self.state.document.settings.env
        except AttributeError:
            # No Sphinx environment, return path as-is
            return filename

        # Check if the path is absolute (starts with / on Unix or drive letter on Windows)
        is_absolute = os.path.isabs(filename)

        if is_absolute:
            # Treat absolute path as relative to source directory
            # Strip the leading slash/drive and make it relative
            if filename.startswith('/'):
                # Unix-style absolute path
                relative_path = filename.lstrip('/')
            elif len(filename) >= 2 and filename[1] == ':':
                # Windows-style absolute path (e.g., C:\path)
                relative_path = filename[2:].lstrip('\\/')
            else:
                relative_path = filename.lstrip('/\\')

            # Get the source directory
            srcdir = env.srcdir

            # Construct the full path from source directory
            full_path = os.path.join(srcdir, relative_path)

            # Get the current document's directory
            docdir = os.path.dirname(env.doc2path(env.docname))

            # Make the path relative to the current document
            try:
                # Try to make it relative to the document directory
                resolved = os.path.relpath(full_path, docdir)
            except ValueError:
                # On Windows, if paths are on different drives, can't make relative
                resolved = relative_path

            # Issue a warning about the path normalization
            logger.warning(
                'csv-table :file: path %s is being interpreted as relative to '
                'the source directory (resolved to %s).',
                filename, resolved,
                location=self.state_machine.get_source_and_line(self.lineno)
            )

            return resolved

        # Relative path - return as-is
        return filename

    def run(self) -> List[nodes.Node]:
        # Handle the file option if present
        if 'file' in self.options:
            original_file = self.options['file']
            resolved_file = self.resolve_file_path(original_file)
            self.options['file'] = resolved_file

        return super().run()

    def make_title(self) -> Tuple[nodes.title, List[system_message]]:
        title, message = super().make_title()
        if title:
            set_source_info(self, title)
        return title, message


def setup(app):
    """Setup the patched directives."""
    app.add_directive('csv-table', CSVTable, override=True)
    return {
        'version': '0.1',
        'parallel_read_safe': True,
    }
