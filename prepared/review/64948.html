```python
# /app/generate_dataset.py
import os
import numpy as np
from PIL import Image, ImageDraw

def ensure_dirs():
    os.makedirs('data/ground_truth', exist_ok=True)
    os.makedirs('data/predictions', exist_ok=True)
    os.makedirs('data/originals', exist_ok=True)

def create_mask(size=(128, 128), shapes=None):
    # shapes: list of (type, bbox) e.g., ('rect', (x1,y1,x2,y2))
    mask = np.zeros(size, dtype=np.uint8)
    if shapes:
        im = Image.fromarray(mask)
        draw = ImageDraw.Draw(im)
        for s in shapes:
            t = s[0]
            if t == 'rect':
                draw.rectangle(s[1], fill=255)
            elif t == 'circle':
                draw.ellipse(s[1], fill=255)
        mask = np.array(im)
    else:
        # simple random foreground
        mask = (np.random.rand(*size) > 0.5).astype(np.uint8) * 255
    return mask

def save_png(arr, path):
    Image.fromarray(arr).save(path)

def generate_scene(i):
    w, h = 128, 128
    # deterministic-ish seed per scene for reproducibility
    rng = np.random.default_rng(seed=100 + i)

    # ground truth: a random blob
    gt_mask = np.zeros((h, w), dtype=np.uint8)
    # draw a random rectangle/ellipse
    x1, y1 = rng.integers(10, 60, size=2)
    x2, y2 = rng.integers(70, 118, size=2)
    gt_shape = rng.choice(['rect', 'circle'])
    if gt_shape == 'rect':
        gt_mask[y1:y2, x1:x2] = 255
    else:
        cx, cy = rng.integers(20, 108, size=2)
        rx, ry = rng.integers(10, 25, size=2)
        for y in range(h):
            for x in range(w):
                if ((x - cx)**2) / (rx**2 + 1e-6) + ((y - cy)**2) / (ry**2 + 1e-6) <= 1:
                    gt_mask[y, x] = 255

    # predicted mask: add some noise/offset
    pred_mask = np.copy(gt_mask)
    # flip a few pixels randomly to simulate imperfect prediction
    mask_flips = rng.integers(0, 200, size=gt_mask.shape[0] * gt_mask.shape[1] // 20)
    for idx in rng.choice(gt_mask.size, size=mask_flips.size, replace=False):
        y = idx // w
        x = idx % w
        pred_mask[y, x] = 255 if pred_mask[y, x] == 0 else 0

    # originals: simple color image
    orig = rng.integers(0, 255, size=(h, w, 3), dtype=np.uint8)
    # draw the ground truth region in the original for a nicer scene
    orig[..., 0] = np.maximum(orig[..., 0], gt_mask // 2)

    return gt_mask, pred_mask, orig

def main():
    ensure_dirs()
    N = 3
    for i in range(1, N+1):
        gt, pred, orig = generate_scene(i)
        gt_path = f'data/ground_truth/scene-{i:02d}.png'
        pred_path = f'data/predictions/scene-{i:02d}.png'
        orig_path = f'data/originals/scene-{i:02d}.png'
        save_png(gt, gt_path)
        save_png(pred, pred_path)
        Image.fromarray(orig).save(orig_path)

if __name__ == "__main__":
    main()
```

```python
# /app/bench_mask.py
import os
import csv
import numpy as np
from PIL import Image
from typing import Tuple

def to_bin255(img_path: str) -> np.ndarray:
    im = Image.open(img_path).convert('L')
    arr = np.array(im, dtype=np.uint8)
    return (arr > 0).astype(np.uint8)

def overlay_mask_on_original(orig_path: str, pred_mask_bin: np.ndarray, out_path: str) -> None:
    orig = Image.open(orig_path).convert('RGB')
    arr = np.array(orig)
    # paint the predicted foreground in red
    red_overlay = np.zeros_like(arr)
    red_overlay[..., 0] = 255  # red channel
    mask = pred_mask_bin[..., None] if pred_mask_bin.ndim == 3 else pred_mask_bin
    # broadcast mask to color channels
    overlay_region = (mask > 0)
    arr_overlay = arr.copy()
    arr_overlay[overlay_region] = [255, 0, 0]  # red where prediction is foreground
    Image.fromarray(arr_overlay).save(out_path)

def compute_metrics(gt_bin: np.ndarray, pred_bin: np.ndarray) -> Tuple[float, int, int]:
    intersection = int((gt_bin & pred_bin).sum())
    union = int(gt_bin.sum() + pred_bin.sum() - intersection)
    iou = float(intersection) / float(union) if union > 0 else 0.0
    matches = intersection
    total_pixels = int(gt_bin.sum())
    return iou, matches, total_pixels

def main():
    gt_dir = 'data/ground_truth'
    pred_dir = 'data/predictions'
    orig_dir = 'data/originals'
    vis_dir = 'vis'
    os.makedirs(vis_dir, exist_ok=True)

    # Prepare report with header
    with open('report.csv', 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(['image', 'iou', 'matches', 'total_pixels'])

        # Iterate over ground-truth masks
        for fname in sorted(os.listdir(gt_dir)):
            if not fname.lower().endswith('.png'):
                continue
            base = fname
            gt_path = os.path.join(gt_dir, fname)
            pred_path = os.path.join(pred_dir, fname)
            orig_path = os.path.join(orig_dir, fname)
            overlay_path = os.path.join(vis_dir, f'{fname[:-4]}_overlay.png')

            # Load and binarize
            gt_bin = to_bin255(gt_path)
            pred_bin = to_bin255(pred_path)

            # Compute metrics
            iou, matches, total_pixels = compute_metrics(gt_bin, pred_bin)

            # Save overlay visualization
            overlay_mask_on_original(orig_path, pred_bin, overlay_path)

            # Write CSV row
            writer.writerow([base, f"{iou:.6f}", matches, total_pixels])

if __name__ == "__main__":
    main()
```
