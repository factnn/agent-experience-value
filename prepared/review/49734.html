```python
class ManifestStaticFilesStorage:
    def __init__(self, max_post_process_passes=1):
        self.max_post_process_passes = max_post_process_passes

    def post_process(self, paths, dry_run=False, **options):
        processed_adjustable_paths = []
        # Simulate building a list of paths that could be post-processed
        for original_path in list(paths.keys()):
            processed_adjustable_paths.append(original_path)

        # Restrict to adjustable paths
        paths = {path: paths[path] for path in processed_adjustable_paths}

        # If max_post_process_passes == 0, this loop won't run, and substitutions
        # will be referenced below without being defined -> UnboundLocalError
        for i in range(self.max_post_process_passes):
            substitutions = False
            # In real code, some substitution logic would set substitutions = True
            if substitutions:
                yield (original_path, None, True)

        # Potential bug: substitutions may be undefined here if the loop never ran
        if substitutions:
            yield ("done", "done", True)
```
