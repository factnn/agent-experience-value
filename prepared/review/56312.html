"""
Implementation of the --report-log plugin for pytest.

This plugin writes a line-based JSON log of test session events,
providing a replacement for the deprecated --resultlog option.
"""
import json
import os

import pytest


def pytest_addoption(parser):
    """Add the --report-log command line option."""
    group = parser.getgroup("terminal reporting", "reporting of test execution")
    group.addoption(
        "--report-log",
        action="store",
        metavar="PATH",
        default=None,
        help="path to a line-based JSON log file for test session events",
    )


def pytest_configure(config):
    """Configure the report log plugin if --report-log is provided."""
    report_log = config.option.report_log
    if report_log and not hasattr(config, "workerinput"):
        # Only register the plugin if not running in a slave/remote context
        config._report_log = ReportLogPlugin(config, report_log)
        config.pluginmanager.register(config._report_log, "report-log-plugin")


class ReportLogPlugin:
    """Plugin that writes test session events as JSON lines to a log file."""

    def __init__(self, config, path):
        self.config = config
        self.path = path
        self._file = None

    def pytest_sessionstart(self, session):
        """Handle session start - open the log file and write initial event."""
        # Create parent directories if needed
        os.makedirs(os.path.dirname(os.path.abspath(self.path)), exist_ok=True)

        # Open file with line buffering
        self._file = open(self.path, "w", buffering=1)

        # Write session start event
        event = {
            "pytest_version": pytest.__version__,
            "type": "SessionStart",
        }
        self._write_event(event)

    def pytest_runtest_logreport(self, report):
        """Handle test report events."""
        if self._file is None:
            return

        serialized = self.config.hook.pytest_report_to_serializable(
            config=self.config, report=report
        )
        if serialized is not None:
            self._write_event(serialized)

    def pytest_collectreport(self, report):
        """Handle collect report events."""
        if self._file is None:
            return

        serialized = self.config.hook.pytest_report_to_serializable(
            config=self.config, report=report
        )
        if serialized is not None:
            self._write_event(serialized)

    def pytest_sessionfinish(self, session, exitstatus):
        """Handle session finish - write final event and close file."""
        if self._file is None:
            return

        # Write session finish event
        event = {
            "exitstatus": exitstatus,
            "type": "SessionFinish",
        }
        self._write_event(event)

        # Close the file
        self._file.close()
        self._file = None

    def _write_event(self, event):
        """Write a single event as a JSON line."""
        self._file.write(json.dumps(event) + "\n")
        self._file.flush()

    def pytest_terminal_summary(self, terminalreporter):
        """Add a summary line indicating where the log was written."""
        if self.path and os.path.exists(self.path):
            terminalreporter.write_sep("-", f"report log: {self.path}")

"""
Implementation of the --report-log plugin for pytest.

This plugin writes a line-based JSON log of test session events,
providing a replacement for the deprecated --resultlog option.
"""
import json
import os

import pytest


def pytest_addoption(parser):
    """Add the --report-log command line option."""
    group = parser.getgroup("terminal reporting", "reporting of test execution")
    group.addoption(
        "--report-log",
        action="store",
        metavar="PATH",
        default=None,
        help="path to a line-based JSON log file for test session events",
    )


def pytest_configure(config):
    """Configure the report log plugin if --report-log is provided."""
    report_log = config.option.report_log
    if report_log and not hasattr(config, "workerinput"):
        # Only register the plugin if not running in a slave/remote context
        config._report_log = ReportLogPlugin(config, report_log)
        config.pluginmanager.register(config._report_log, "report-log-plugin")


class ReportLogPlugin:
    """Plugin that writes test session events as JSON lines to a log file."""

    def __init__(self, config, path):
        self.config = config
        self.path = path
        self._file = None

    def pytest_sessionstart(self, session):
        """Handle session start - open the log file and write initial event."""
        # Create parent directories if needed
        os.makedirs(os.path.dirname(os.path.abspath(self.path)), exist_ok=True)

        # Open file with line buffering
        self._file = open(self.path, "w", buffering=1)

        # Write session start event
        event = {
            "pytest_version": pytest.__version__,
            "type": "SessionStart",
        }
        self._write_event(event)

    def pytest_runtest_logreport(self, report):
        """Handle test report events."""
        if self._file is None:
            return

        serialized = self.config.hook.pytest_report_to_serializable(
            config=self.config, report=report
        )
        if serialized is not None:
            self._write_event(serialized)

    def pytest_collectreport(self, report):
        """Handle collect report events."""
        if self._file is None:
            return

        serialized = self.config.hook.pytest_report_to_serializable(
            config=self.config, report=report
        )
        if serialized is not None:
            self._write_event(serialized)

    def pytest_sessionfinish(self, session, exitstatus):
        """Handle session finish - write final event and close file."""
        if self._file is None:
            return

        # Write session finish event
        event = {
            "exitstatus": exitstatus,
            "type": "SessionFinish",
        }
        self._write_event(event)

        # Close the file
        self._file.close()
        self._file = None

    def _write_event(self, event):
        """Write a single event as a JSON line."""
        self._file.write(json.dumps(event) + "\n")
        self._file.flush()

    def pytest_terminal_summary(self, terminalreporter):
        """Add a summary line indicating where the log was written."""
        if self.path and os.path.exists(self.path):
            terminalreporter.write_sep("-", f"report log: {self.path}")
