```python
# app/vending.py
import logging

class VendingMachine:
    # Define five items with prices
    PRODUCTS = {
        'A': {'name': 'Chips', 'price': 1.25},
        'B': {'name': 'Candy', 'price': 1.00},
        'C': {'name': 'Soda',  'price': 1.50},
        'D': {'name': 'Gum',   'price': 0.75},
        'E': {'name': 'Water', 'price': 1.00},
    }

    def __init__(self, stock=None, logger=None):
        # Stock initialized to 5 each by default
        self.stock = stock if stock is not None else {k: 5 for k in self.PRODUCTS}
        self.logger = logger or logging.getLogger(__name__)

    def print_stock(self, out=None):
        lines = []
        for code in sorted(self.PRODUCTS.keys()):
            item = self.PRODUCTS[code]
            lines.append(f"{code}: {item['name']} - ${item['price']:.2f} stock={self.stock.get(code, 0)}")
        text = "\n".join(lines)
        if out:
            out.write(text + "\n")
        else:
            print(text)
        return text

    def purchase(self, code, amount_paid):
        if code not in self.PRODUCTS:
            self.logger.error(f"Invalid product code: {code}")
            return {'status': 'error', 'reason': 'invalid_code'}

        if self.stock.get(code, 0) <= 0:
            self.logger.info(f"Out of stock for {code}")
            return {'status': 'fail', 'reason': 'out_of_stock', 'change': amount_paid}

        price = self.PRODUCTS[code]['price']
        change = amount_paid - price
        if change < 0:
            self.logger.info(f"Insufficient funds for {code}: paid {amount_paid}, price {price}")
            return {'status': 'fail', 'reason': 'insufficient_funds', 'change': amount_paid}

        self.stock[code] -= 1
        self.logger.info(f"Purchased {code} for ${price:.2f}, change ${change:.2f}, stock now {self.stock[code]}")
        return {'status': 'ok', 'code': code, 'price': price, 'change': change}

    def simulate_purchases(self, sequence):
        # sequence: list of (code, amount_paid)
        results = []
        for code, amount in sequence:
            res = self.purchase(code, amount)
            results.append((code, amount, res))
        return results
```

```python
# app/env_config.py
import configparser
import os

def load_config(path="config/environments.ini"):
    parser = configparser.ConfigParser()
    if not os.path.exists(path):
        return None
    parser.read(path)
    return parser

def get_vpc_by_name(name, parser=None):
    if parser is None:
        parser = load_config()
    if not parser:
        return None

    section = f"vpc:{name}"
    if section in parser:
        cfg = parser[section]
        vpc_id = cfg.get('id')
        region = cfg.get('region')
        if vpc_id and region:
            return {'name': name, 'id': vpc_id, 'region': region}

    # Fallback: search any section that starts with "vpc:" and matches the name
    for sec in parser.sections():
        if sec.lower().startswith("vpc:"):
            if sec.split(":", 1)[1] == name:
                cfg = parser[sec]
                vpc_id = cfg.get('id')
                region = cfg.get('region')
                if vpc_id and region:
                    return {'name': name, 'id': vpc_id, 'region': region}
    return None
```

```python
# app/tests/test_vending_and_config.py
from vending import VendingMachine
from env_config import load_config, get_vpc_by_name

def test_stock_decreases_after_purchase():
    vm = VendingMachine()
    initial = vm.stock['A']
    res = vm.purchase('A', 2.0)  # price 1.25
    assert res['status'] == 'ok'
    assert vm.stock['A'] == initial - 1

def test_out_of_stock_handling():
    vm = VendingMachine(stock={'A': 0, 'B': 5, 'C': 5, 'D': 5, 'E': 5})
    res = vm.purchase('A', 2.0)
    assert res['status'] == 'fail' and res['reason'] == 'out_of_stock'

def test_non_negative_change():
    vm = VendingMachine()
    # valid purchase should yield non-negative change
    res = vm.purchase('B', 2.0)  # price 1.00
    assert res['status'] == 'ok'
    assert res['change'] >= 0

def test_vpc_loading():
    parser = load_config('config/environments.ini')
    vpc = get_vpc_by_name('corp-vpc', parser)
    assert vpc is not None
    assert vpc['id'] == 'vpc-corp-123'
    assert vpc['region'] == 'us-east-1'
```

```bash
#!/usr/bin/env bash
set -euo pipefail

LOG_DIR="logs"
LOG_FILE="$LOG_DIR/vmbench.log"

# Clear prior logs and ensure log directory exists
rm -f "$LOG_FILE"
mkdir -p "$LOG_DIR"

echo "Starting deterministic benchmark..." | tee -a "$LOG_FILE"

python3 - <<'PY' >> "$LOG_FILE" 2>&1
from vending import VendingMachine

vm = VendingMachine()
sequence = [
    ('A', 2.00),  # should succeed, stock reduces
    ('B', 1.00),  # exact price, stock reduces
    ('C', 2.00),  # overpay, gets change
    ('D', 0.50),  # insufficient funds
    ('E', 2.00),   # overpay
]

for code, paid in sequence:
    res = vm.purchase(code, paid)
    print(f"PURCHASE {code} paid={paid} -> {res}")

print("FINAL STOCK:", vm.stock)
PY

echo "Benchmark finished." | tee -a "$LOG_FILE"
```
