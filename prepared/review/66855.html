```python
# app/env.py - before state
class MarioGridEnv:
    def __init__(self):
        self.size = 5
        self.pos = (0, 0)
        self.max_steps = 12
        self.steps = 0

    def reset(self):
        self.pos = (0, 0)
        self.steps = 0
        return {'position': self.pos, 'reward': 0, 'done': False, 'goal_achieved': False}

    def step(self, action):
        x, y = self.pos
        if action == 0:
            y = max(0, y - 1)
        elif action == 1:
            y = min(self.size - 1, y + 1)
        elif action == 2:
            x = min(self.size - 1, x + 1)
        elif action == 3:
            x = max(0, x - 1)
        self.pos = (x, y)
        self.steps += 1
        done = self.steps >= self.max_steps
        return {'position': self.pos, 'reward': 0, 'done': done, 'goal_achieved': False}
```
