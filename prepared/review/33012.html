#!/bin/bash

# Script to identify/remove files with modification times during night hours (22:45-05:00)
# Usage: ./filter_night_photos.sh [directory] [--delete]

DIR="${1:-.}"
DELETE=false

if [ "$2" = "--delete" ]; then
    DELETE=true
fi

# Function to check if a time (HH:MM) is in night range (22:45-05:00)
is_night_time() {
    local time="$1"
    local hour=${time%:*}
    local minute=${time#*:}
    local total_minutes=$((hour * 60 + minute))

    # Night: 22:45 (1365 min) to 05:00 (300 min)
    # This spans midnight, so we check if it's either:
    # - After 22:45 OR
    # - Before 05:00
    [ $total_minutes -ge 1365 ] || [ $total_minutes -lt 300 ]
}

# Process files
count=0
deleted_count=0

for file in "$DIR"/*; do
    [ -f "$file" ] || continue

    # Get modification time in HH:MM format
    mtime=$(stat -c %y "$file" | cut -d' ' -f2 | cut -d: -f1,2)

    if is_night_time "$mtime"; then
        echo "Nighttime file: $file (mtime: $mtime)"
        count=$((count + 1))
        if $DELETE; then
            rm "$file"
            deleted_count=$((deleted_count + 1))
        fi
    fi
done

echo ""
echo "Found $count nighttime files"
if $DELETE; then
    echo "Deleted $deleted_count files"
else
    echo "Run with --delete flag to remove them"
fi

# Solutions to Remove Nighttime Photos (22:45-05:00)

## Solution 1: Bash Script (Recommended)

The `filter_night_photos.sh` script provides a safe, easy-to-use solution.

Usage:

# First, list what will be deleted
./filter_night_photos.sh /path/to/photos

# Actually delete the files

## Solution 2: One-liner using find with stat

Find and remove nighttime photos:

find /path/to/photos -type f -exec sh -c '
  mtime=$(stat -c %y "$1" | cut -d" " -f2 | cut -d: -f1,2);
  hour=${mtime%:*}; minute=${mtime#*:};
  mins=$((hour*60 + minute));
  [ $mins -ge 1365 ] || [ $mins -lt 300 ]
' sh {} \; -print -delete


## Solution 3: Using touch + find (more portable)

Create reference files for the night boundaries:

# Create reference files for 22:45 and 05:00 on a specific date
touch -d "2024-01-01 22:45:00" /tmp/night_start
touch -d "2024-01-02 05:00:00" /tmp/night_end

# Find files in the first night window (22:45 to midnight)
find /path/to/photos -type f -newermt "2024-01-01 22:45" ! -newermt "2024-01-02" -print

# You'd need to loop through all dates for this approach


## Solution 4: Perl one-liner


perl -e '
  use File::Find;
  find(sub {
    return unless -f;
    my @stat = stat($_);
    my ($sec,$min,$hour) = localtime($stat[9]);
    my $mins = $hour*60 + $min;
    if ($mins >= 1365 || $mins < 300) {
      print "$File::Find::name\n";
      # unlink($_);  # Uncomment to delete
    }
  }, "/path/to/photos");
'


## Time Calculations

- 22:45 = 22 × 60 + 45 = 1365 minutes
- 05:00 = 5 × 60 = 300 minutes

Because the night spans midnight, we check: time >= 22:45 OR time < 05:00
