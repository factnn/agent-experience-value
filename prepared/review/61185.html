// Parse command line arguments
function parseArgs() {
  const args = process.argv.slice(2);
  const result = {};

  for (let i = 0; i < args.length; i++) {
    const arg = args[i];
    if (arg.startsWith('--')) {
      const key = arg.slice(2);
      const value = args[i + 1];
      if (value && !value.startsWith('--')) {
        if (key === 'page' || key === 'limit') {
          result[key] = parseInt(value, 10);
        } else {
          result[key] = value;
        }
        i++;
      } else {
        result[key] = true;
      }
    }
  }

  return result;
}

function main() {
  const args = parseArgs();
  const { proposalId, page = 1, limit = 10, option } = args;

  if (!proposalId) {
    console.error('Error: --proposalId is required');
    process.exit(1);

  const proposalsPath = path.join(__dirname, '..', 'data', 'proposals.json');

  // Find proposal by id
  const proposal = proposals.find(p => p.id === proposalId);
  if (!proposal) {
    console.error(`Error: Proposal with id "${proposalId}" not found`);
    process.exit(1);
  }

  // Load accounts
  const accountsPath = path.join(__dirname, '..', 'data', 'accounts.json');
  const accounts = JSON.parse(fs.readFileSync(accountsPath, 'utf8'));

  // Get votes from proposal
  const votes = proposal.votes || [];

  // If no votes, return empty result
  if (votes.length === 0) {
    const result = {

  // Deduplicate votes by voter scanning from the end (latest wins)
  const voterMap = new Map();
  for (let i = votes.length - 1; i >= 0; i--) {

      voterMap.set(vote.voter, vote);

  // Total count after deduplication and filtering

  // Apply pagination

    page: page,
    limit: limit,
