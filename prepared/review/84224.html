"""
Fix for pytezos counter mismatch error

The error occurs when the local counter state is out of sync with the blockchain.
This script demonstrates multiple ways to resolve the issue.
"""

from pytezos import pytezos

# Your private key (replace with your actual key)
PRIVATE_KEY = "your_private_key_here"

# Your contract address (from the error message)
CONTRACT_ADDRESS = "tz1YY1LvD6TFH4z74pvxPQXBjAKHE5tB5Q8f"

# RPC node (you can change this)
RPC_URL = "https://mainnet.tezrpc.me"

def method1_sync_counter_from_node():
    """
    Method 1: Get the current counter from the node and use it directly
    """
    print("Method 1: Syncing counter from node...")

    pytezos_client = pytezos.using(key=PRIVATE_KEY, shell=RPC_URL)

    # Get the actual counter from the blockchain
    actual_counter = pytezos_client.contract(CONTRACT_ADDRESS).counter()
    print(f"Current counter on chain: {actual_counter}")

    # Create operation with the correct counter
    # The counter should be actual_counter + 1
    op = pytezos_client.transaction(
        destination=CONTRACT_ADDRESS,
        amount=0,
        counter=actual_counter + 1
    )

    return op

def method2_reset_counter():
    """
    Method 2: Create a fresh client and let pytezos handle counter
    """
    print("Method 2: Creating fresh client...")

    # Create a new client instance - this resets cached state
    pytezos_client = pytezos.using(key=PRIVATE_KEY, shell=RPC_URL)

    # For deployment, use the account's counter
    account_counter = pytezos_client.account()['counter']
    print(f"Account counter: {account_counter}")

    return pytezos_client

    Method 4: Using context to get the correct counter
