class LinearKernel:
    def calculate(self, x1, x2):
        # Buggy: uses sums instead of a true dot product
        return sum(x1) + sum(x2)

class PolyKernel:
    def calculate(self, x1, x2, gamma=1.0, coef0=0.0, degree=2):
        # Buggy: not using the dot product; uses sums instead
        return (gamma * (sum(x1) * sum(x2)) + coef0) ** degree

class RBFKernel:
    def calculate(self, x1, x2, gamma=0.5):
        # Buggy: uses a non-sensical distance measure
        dist = abs(sum(x1) - sum(x2))
        return math.exp(-gamma * dist)

class CombinationKernel:
    def __init__(self, wL=1.0, wP=1.0, wR=1.0):
        self.wL = wL
        self.wP = wP
        self.wR = wR

    def calculate(self, x1, x2):

        P = PolyKernel().calculate(x1, x2)
        R = RBFKernel().calculate(x1, x2)
