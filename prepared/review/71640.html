1) preprocess.py (before)
```python
import glob
import os
import json
import pandas as pd

DATA_DIR = "data"
OUTPUT_DIR = "output"

def discover_common_columns(files):
    common = None
    for f in files:
        df = pd.read_csv(f)
        cols = set(df.columns)
        if common is None:
            common = cols
        else:
            common &= cols
    return sorted(list(common)) if common else []

def write_common_columns(path, cols):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w") as f:
        json.dump(cols, f)

def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    csvs = sorted(glob.glob(os.path.join(DATA_DIR, "shiptracks_*.csv")))
    common = discover_common_columns(csvs)
    write_common_columns(os.path.join(DATA_DIR, "common_columns.json"), common)

    if not csvs:
        print("No input files found.")
        return

    frames = []
    for f in csvs:
        df = pd.read_csv(f)
        # use all columns found (even if not common)
        frames.append(df)
    data = pd.concat(frames, ignore_index=True)

    # Very naive: assume columns dataset_id, time, longitude, latitude exist
    dataset_ids = data["dataset_id"].unique() if "dataset_id" in data.columns else []
    for ds in dataset_ids:
        subset = data[data["dataset_id"] == ds]
        if "time" not in subset.columns:
            continue
        subset["time"] = pd.to_datetime(subset["time"])
        subset.set_index("time", inplace=True)
        hourly = subset[["longitude", "latitude"]].resample("1H").mean().dropna()
        out_dir = os.path.join(OUTPUT_DIR, "hourly")
        os.makedirs(out_dir, exist_ok=True)
        hourly.to_csv(os.path.join(out_dir, f"{ds}.csv"))

    # Combined
    all_hourly = []
    for f in glob.glob(os.path.join(OUTPUT_DIR, "hourly", "*.csv")):
        ds = os.path.basename(f).split(".")[0]
        tmp = pd.read_csv(f)
        tmp["dataset_id"] = ds
        all_hourly.append(tmp)
    if all_hourly:
        combined = pd.concat(all_hourly, ignore_index=True)
        combined.to_csv(os.path.join(OUTPUT_DIR, "combined_hourly.csv"), index=False)

    print("Processing complete.")

if __name__ == "__main__":
    main()
```

import pandas as pd
import numpy as np
from datetime import datetime, timedelta

DATA_DIR = "data"
os.makedirs(DATA_DIR, exist_ok=True)

def generate_shiptrack(name, start_time, n=24):
    times = [start_time + timedelta(hours=i) for i in range(n)]
    lon = np.cumsum(np.random.randn(n)) + (-70)
    lat = np.cumsum(np.random.randn(n)) + 40
    df = pd.DataFrame({

        "time": times,
        "longitude": lon,
        "latitude": lat,
        "other_col": np.random.randint(0, 100, size=n)
    })
    path = os.path.join(DATA_DIR, f"shiptracks_{name}.csv")
    df.to_csv(path, index=False)

def main():
    now = datetime.utcnow()
    generate_shiptrack("A", now - timedelta(days=1))
    generate_shiptrack("B", now - timedelta(days=1, hours=6))

if __name__ == "__main__":
