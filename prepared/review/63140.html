- silabify_cli.py
```python
#!/usr/bin/env python3
import sys
from typing import Dict

# Deterministic mapping for the provided tiny corpus
SPLIT_MAP: Dict[str, str] = {
    "casa": "ca-sa",
    "amigo": "a-mi-go",
    "cidade": "ci-da-de",
    "vento": "ven-to",
    "hospital": "hos-pi-tal",
    "banana": "ba-na-na",
    "música": "mú-si-ca",
    "escola": "es-co-la",
    "caminho": "ca-mi-nho",
    "coração": "co-ra-ção",
}

VOWELS = set("aeiouáéíóúâêãõà")

def syllabify_word(word: str) -> str:
    w = word.strip()
    if not w:
        return ""
    if w in SPLIT_MAP:
        return SPLIT_MAP[w]
    # Simple fallback: split before each vowel after the first character
    parts = []
    start = 0
    for i in range(1, len(w)):
        if w[i] in VOWELS and w[i-1] not in VOWELS:
            parts.append(w[start:i])
            start = i
    parts.append(w[start:])
    return "-".join(parts)

def main():
    # Read input: file path as arg or stdin
    words = []
    if len(sys.argv) > 1:
        path = sys.argv[1]
        with open(path, "r", encoding="utf-8") as f:
            words = [line.strip() for line in f if line.strip()]
    else:
        words = [line.strip() for line in sys.stdin if line.strip()]

    for w in words:
        print(f"{w}\t{syllabify_word(w)}")

if __name__ == "__main__":
    main()
```

```bash
#!/bin/bash
set -euo pipefail

OUTPUT="timing.csv"
echo "size,time_s" > "$OUTPUT"

# Helper: create a temp file with first N words
TMP=$(mktemp)
trap "rm -f $TMP" EXIT

for SIZE in 1 5 10; do
  head -n "$SIZE" words.txt > "$TMP"
  # Run the CLI and measure wall-clock time
  /usr/bin/time -f "%e" -o /tmp/bench_time.$$ ./silabify_cli.py "$TMP" > /dev/null 2>&1
  TIME=$(cat /tmp/bench_time.$$)
  echo "$SIZE,$TIME" >> "$OUTPUT"
done
```

- verify.py
```python
#!/usr/bin/env python3
import sys

def main():
    if len(sys.argv) != 3:
        print("Usage: verify.py actual.txt expected.txt")
        sys.exit(2)
    actual_path, expected_path = sys.argv[1], sys.argv[2]
    with open(actual_path, "r", encoding="utf-8") as a, \
         open(expected_path, "r", encoding="utf-8") as e:
        actual_lines = [ln.rstrip("\n") for ln in a]
        expected_lines = [ln.rstrip("\n") for ln in e]

    if len(actual_lines) != len(expected_lines):
        print(f"Length mismatch: actual {len(actual_lines)} lines vs expected {len(expected_lines)} lines")
        sys.exit(1)

    for i, (al, el) in enumerate(zip(actual_lines, expected_lines), start=1):
        if al != el:
            print(f"Mismatch at line {i}: actual='{al}' expected='{el}'")
            sys.exit(1)

    print("OK")
    sys.exit(0)

if __name__ == "__main__":
    main()
```
