#!/usr/bin/env python3
"""Process CSV files, collate data, generate plot, and write log."""
import os
import glob
import csv
from datetime import datetime
import pandas as pd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

WORKSPACE = "bench-ZZZ"
EXTRACTED_DIR = os.path.join(WORKSPACE, "extracted")
MASTER_CSV = os.path.join(WORKSPACE, "master.csv")
PLOT_FILE = os.path.join(WORKSPACE, "plot.png")
LOG_DIR = os.path.join(WORKSPACE, "logs")
LOG_FILE = os.path.join(LOG_DIR, "log.txt")

def process_csvs():
    """Process all CSV files, skip 8 header rows, collate data."""
    # Find all matching CSV files
    pattern = os.path.join(EXTRACTED_DIR, "Occupancy_Monitor_Client_Count_-_Most_Buildings_*.csv")
    csv_files = glob.glob(pattern)

    if not csv_files:
        raise FileNotFoundError(f"No CSV files found matching pattern: {pattern}")

    # Collect all data rows
    header = ["Key", "Event Time", "Associated Client Count", "Authenticated Client Count"]
    all_rows = []

    for csv_file in csv_files:
        with open(csv_file, 'r') as f:
            # Skip first 8 rows (header lines)
            for _ in range(8):
                next(f, None)

            # Read the actual header row (row 9)
            actual_header = f.readline().strip().split(',')

            # Read data rows
            reader = csv.reader(f)
            for row in reader:
                if row and len(row) >= 4:  # Ensure we have enough columns
                    all_rows.append(row)

    # Write master CSV
    with open(MASTER_CSV, 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(header)
        writer.writerows(all_rows)

    return len(csv_files), len(all_rows)

def generate_plot():
    """Generate plot of Authenticated Client Count vs Event Time."""
    df = pd.read_csv(MASTER_CSV)

    # Parse Event Time
    df['Event Time'] = pd.to_datetime(df['Event Time'])

    # Sort by time
    df = df.sort_values('Event Time')

    # Create plot
    plt.figure(figsize=(10, 6))
    plt.plot(df['Event Time'], df['Authenticated Client Count'], marker='o', linestyle='-')
    plt.xlabel('Event Time')
    plt.ylabel('Authenticated Client Count')
    plt.title('Authenticated Client Count vs Event Time')
    plt.xticks(rotation=45)
    plt.tight_layout()
    plt.savefig(PLOT_FILE)
    plt.close()

def write_log(num_csvs, num_rows):
    """Write log summary."""
    os.makedirs(LOG_DIR, exist_ok=True)

    with open(LOG_FILE, 'w') as f:
        f.write(f"Workflow Summary\n")
        f.write(f"="*40 + "\n")
        f.write(f"Number of input CSVs processed: {num_csvs}\n")
        f.write(f"Total data rows written to master.csv: {num_rows}\n")
        f.write(f"Path to artifacts: {os.path.abspath(WORKSPACE)}\n")
        f.write(f"\nGenerated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")

def main():
    """Main processing function."""
    print("Processing CSV files...")
    num_csvs, num_rows = process_csvs()
    print(f"Processed {num_csvs} CSV files with {num_rows} data rows")

    print("Generating plot...")
    generate_plot()
    print(f"Plot saved to {PLOT_FILE}")

    print("Writing log...")
    write_log(num_csvs, num_rows)
    print(f"Log saved to {LOG_FILE}")

    print("\nWorkflow complete!")
    print(f"Artifacts in: {WORKSPACE}/")

if __name__ == "__main__":
    main()
