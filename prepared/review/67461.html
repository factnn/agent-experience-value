2) app/scripts/run_pipeline.py (before)
```python
#!/usr/bin/env python3
import json
import pandas as pd
import numpy as np
from pathlib import Path

DATA_DIR = Path('data')

def load_datasets():
    train = pd.read_parquet(DATA_DIR / 'train_preprocessed.parquet')
    test = pd.read_parquet(DATA_DIR / 'test_preprocessed.parquet')
    with open(DATA_DIR / 'itemid_map.json', 'r') as f:
        mapping = json.load(f)
    return train, test, mapping

def main():
    train, test, mapping = load_datasets()

    # Map references to indices (will raise KeyError if unmapped)
    train['reference_idx'] = train['reference'].apply(lambda r: mapping[r])
    test['reference_idx'] = test['reference'].apply(lambda r: mapping[r])

    # Compute per-session offsets
    def compute_offsets(df):
        offsets = [0]
        total = 0
        for sid, grp in df.groupby('session_id', sort=True):
            total += len(grp)
            offsets.append(total)
        return np.asarray(offsets, dtype=np.int32)

    offsets_train = compute_offsets(train)
    offsets_test = compute_offsets(test)

    np.save(DATA_DIR / 'offset_sessions_train.npy', offsets_train)
    np.save(DATA_DIR / 'offset_sessions_test.npy', offsets_test)

if __name__ == '__main__':
    main()
```
