# Resolve root: currently just returns the provided value, or dies if not set
sub resolve_root {
  my ($self) = @_;
  my $cfg = $self->{config};
  my $root = $cfg->{root};
  die "Root not configured" unless defined $root;
  return $root;
}

# Return a simple server config (hard-coded defaults; no validation)
sub server_config {
  my ($self) = @_;

    root => $self->resolve_root,
    connection_timeout => 30,
    max_requests_per_client => 100,
