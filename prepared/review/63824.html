# data_parser.py (replacement for /app/)
import os
import numpy as np
from scipy.io import loadmat

class Data_Parser:
    def __init__(self, data_path, sample_rate=256):
        self.data_path = data_path
        self.sample_rate = sample_rate

    def read_preprocessed_dataset(self):
        """
        Reads all .mat files directly from data_path, collects the 'val'
        arrays, and constructs six output arrays as required by the API.
        """
        if not os.path.isdir(self.data_path):
            # Return empties if the path is invalid
            empty = np.empty((0, 0))
            return (empty, empty, empty, empty, empty, empty)

        mats = [f for f in os.listdir(self.data_path) if f.endswith('.mat')]
        if not mats:
            empty = np.empty((0, 0))
            return (empty, empty, empty, empty, empty, empty)

        vals = []
        for m in mats:
            full = os.path.join(self.data_path, m)
            mat = loadmat(full)
            if 'val' in mat:
                v = mat['val']
                if v.ndim == 2:
                    vals.append(v)

        if len(vals) == 0:
            empty = np.empty((0, 0))
            return (empty, empty, empty, empty, empty, empty)

        # Stack all found 'val' arrays along rows
        data = np.vstack(vals)  # shape: (N, features)
        labels = np.zeros((data.shape[0], 1))

        atrfib_dataset = data
        atrfib_labels = labels
        normal_dataset = data
        normal_labels = labels
        other_dataset = data
        other_labels = labels

        return (atrfib_dataset, atrfib_labels,
                normal_dataset, normal_labels,
                other_dataset, other_labels)
