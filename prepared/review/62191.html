from django.apps import apps
from django.contrib.contenttypes.models import ContentType


def get_contenttypes_and_models(app_config, db, ContentType):
    """
    Return a tuple of (content_types_dict, app_models_set).

    content_types_dict: Mapping of model_name -> ContentType for this app
    app_models_set: Set of model names that exist in the app
    """
    content_types = ContentType.objects.using(db).filter(
        app_label=app_config.label
    )
    content_types_dict = {ct.model: ct for ct in content_types}

    # Get all models in this app
    app_models = set()
    for model in app_config.get_models():
        app_models.add(model._meta.model_name)

    return content_types_dict, app_models
