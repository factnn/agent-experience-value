"""
This module contains the Options class for Django model _meta.
"""

import copy
from django.db.models import Q
from django.db.models.constraints import BaseConstraint
from django.db.models.indexes import Index


class Options:
    def __init__(self, meta=None):
        self.abstract = False
        self.constraints = []
        self.indexes = []
        self.unique_together = []
        self.index_together = []
        self.app_label = None

        if meta:
            for attr in dir(meta):
                if not attr.startswith('_'):
                    value = getattr(meta, attr)
                    if not callable(value):
                        setattr(self, attr, value)

    def contribute_to_class(self, cls, name):
        """
        Add necessary information to the class.
        """
        self.unique_together = self._normalize_together(self.unique_together)
        self.index_together = self._normalize_together(self.index_together)

        # Interpolate %(app_label)s and %(class)s in constraint and index names
        # only for non-abstract models
        if not self.abstract:
            self.constraints = self._format_names_with_class(cls, self.constraints)
            self.indexes = self._format_names_with_class(cls, self.indexes)

    def _normalize_together(self, together):
        """
        Normalize together options.
        """
        if not together:
            return []
        if isinstance(together, (list, tuple)):
            return [tuple(fields) for fields in together]
        return []

    def _format_names_with_class(self, cls, objs):
        """
        Format constraint/index names with app_label and class placeholders.

        Replaces %(app_label)s with the lowercased app label and
        %(class)s with the lowercased class name.
        """
        if not objs:
            return []

        formatted_objs = []
        format_dict = {
            'app_label': cls._meta.app_label.lower() if cls._meta.app_label else '',
            'class': cls.__name__.lower(),
        }

        for obj in objs:
            # Clone the object to avoid modifying the original
            new_obj = copy.copy(obj)
            if hasattr(obj, 'name') and obj.name:
                try:
                    new_obj.name = obj.name % format_dict
                except (KeyError, TypeError):
                    # If formatting fails, keep the original name
                    new_obj.name = obj.name
            formatted_objs.append(new_obj)

        return formatted_objs
