```python
import os
import json
import subprocess
import time
import signal

def read_basic_infos():
    with open("BasicInfos.conf", "r") as f:
        line = f.readline().strip()
    # Expect: multusModulesConfig=/path/to/modules.json
    key, val = line.split("=", 1)
    return val

def load_module_config(modules_json_path):
    with open(modules_json_path, "r") as f:
        data = json.load(f)
    return data

def start_module(module):
    binary_path = module["ModuleBinaryPath"]
    log_path = module.get("ModuleLogFile", "/tmp/multusd_demo/logs/default.log")
    pid_path = module["ModulePIDFile"]

    log_file = open(log_path, "a")
    proc = subprocess.Popen([binary_path], stdout=log_file, stderr=log_file)
    with open(pid_path, "w") as pf:
        pf.write(str(proc.pid))
    return proc

def main():
    modules_json_path = read_basic_infos()
    module = load_module_config(modules_json_path)
    proc = start_module(module)

    # Simple loop to keep the supervisor alive
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        try:
            proc.terminate()
        except Exception:
            pass
        print("Supervisor exiting")

if __name__ == "__main__":
    main()
```

#!/usr/bin/env python3
import os
import json
import subprocess
import time
import signal

def log_event(module, message):
    """Log an event to the module's log file."""
    log_path = module.get("ModuleLogFile", "/tmp/multusd_demo/logs/default.log")
    timestamp = time.strftime("%Y-%m-%d %H:%M:%S")
    with open(log_path, "a") as f:
        f.write(f"{timestamp} [SUPERVISOR] {message}\n")

def read_basic_infos():
    """Read BasicInfos.conf to get the modules.json path."""
    with open("BasicInfos.conf", "r") as f:
        line = f.readline().strip()
    # Expect: multusModulesConfig=/path/to/modules.json
    key, val = line.split("=", 1)
    return val

def load_module_config(modules_json_path):
    """Load the module configuration from JSON."""
    with open(modules_json_path, "r") as f:
        data = json.load(f)
    return data

def get_update_trigger_mtime(module):
    """Get the modification time of the update.trigger file, or None if not exists."""
    update_dir = module["ModuleUpdateDir"]
    trigger_path = os.path.join(update_dir, "update.trigger")
    if os.path.exists(trigger_path):
        return os.path.getmtime(trigger_path)
    return None

def start_module(module):
    """Start the module binary and return the process."""
    binary_path = module["ModuleBinaryPath"]
    log_path = module.get("ModuleLogFile", "/tmp/multusd_demo/logs/default.log")
    pid_path = module["ModulePIDFile"]

    # Ensure log directory exists
    os.makedirs(os.path.dirname(log_path), exist_ok=True)
    # Ensure pid directory exists
    os.makedirs(os.path.dirname(pid_path), exist_ok=True)

    log_event(module, f"Starting module: {binary_path}")

    log_file = open(log_path, "a")
    proc = subprocess.Popen([binary_path], stdout=log_file, stderr=log_file)

    with open(pid_path, "w") as pf:
        pf.write(str(proc.pid))

    log_event(module, f"Module started with PID: {proc.pid}")
    return proc

def stop_module(proc, module):
    """Stop the running module process."""
    try:
        proc.terminate()
        log_event(module, f"Terminated module process (PID: {proc.pid})")
    except Exception as e:
        log_event(module, f"Error stopping module: {e}")

def main():
    # Read configuration
    modules_json_path = read_basic_infos()
    module = load_module_config(modules_json_path)

    # Track the last modification time of update.trigger
    last_trigger_mtime = get_update_trigger_mtime(module)

    # Start the module
    proc = start_module(module)

    log_event(module, "Supervisor started, monitoring for updates...")

    # Main loop with update monitoring
    try:
        while True:
            time.sleep(0.5)  # Check for updates every 0.5 seconds

            # Check if update.trigger has been created or modified
            current_trigger_mtime = get_update_trigger_mtime(module)

            if current_trigger_mtime is not None:
                if last_trigger_mtime is None or current_trigger_mtime > last_trigger_mtime:
                    log_event(module, "Update trigger detected! Restarting module...")

                    # Stop the old process
                    stop_module(proc, module)

                    # Start a new process
                    proc = start_module(module)

                    # Update the last known modification time
                    last_trigger_mtime = current_trigger_mtime

    except KeyboardInterrupt:
        log_event(module, "Supervisor received interrupt, shutting down...")
        try:
            stop_module(proc, module)
        except Exception:
            pass
        print("Supervisor exiting")

if __name__ == "__main__":
    main()
