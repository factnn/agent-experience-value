2) app/generate_synthetic.py (new)
- This script deterministically generates three synthetic Gaussian blobs on the same 120x200 canvas, centered at (y=60, x=100). It saves three color PNGs:
  - synthetic_XRay.png (red channel)
  - synthetic_Radio.png (green channel)
  - synthetic_GammaRay.png (blue channel)

```python
# app/generate_synthetic.py
import numpy as np
import imageio

def gaussian_2d(shape, center, sigma):
    h, w = shape
    y = np.linspace(0, h - 1, h)
    x = np.linspace(0, w - 1, w)
    X, Y = np.meshgrid(x, y)
    cx, cy = center
    sx, sy = sigma
    # 2D Gaussian (anisotropic)
    g = np.exp(-(((X - cx) ** 2) / (2.0 * sx * sx) + ((Y - cy) ** 2) / (2.0 * sy * sy)))
    g = g / g.max()
    return g

def save_png_band(shape, center, sigma, color_index, path, scale=1.0):
    # color_index: 0->red, 1->green, 2->blue
    g = gaussian_2d(shape, center, sigma)
    # prepare 3-channel image with only the chosen channel populated
    img = np.zeros((shape[0], shape[1], 3), dtype=np.float32)
    img[..., color_index] = g * scale
    # clip and convert to uint8
    img = (np.clip(img, 0.0, 1.0) * 255).astype(np.uint8)
    imageio.imwrite(path, img)

def main():
    height, width = 120, 200
    center = (100, 60)  # (x, y) for gaussian generator center
    sigma = (18.0, 18.0)

    # XRay (red channel)
    save_png_band((height, width), center, sigma, color_index=0, path='synthetic_XRay.png', scale=1.0)
    # Radio (green channel)
    save_png_band((height, width), center, sigma, color_index=1, path='synthetic_Radio.png', scale=1.0)
    # GammaRay (blue channel)
    save_png_band((height, width), center, sigma, color_index=2, path='synthetic_GammaRay.png', scale=1.0)

    print("Synthetic bands generated: synthetic_XRay.png, synthetic_Radio.png, synthetic_GammaRay.png")

if __name__ == "__main__":
    main()
```
